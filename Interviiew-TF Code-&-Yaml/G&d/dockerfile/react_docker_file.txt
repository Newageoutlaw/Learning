# ===== Stage 1: Build =====
FROM node:18-alpine AS builder      # Node.js lightweight image for build
LABEL maintainer="dharmpal.singh@example.com"
WORKDIR /app                        # Working directory inside container
COPY package*.json ./               # Copy package.json & package-lock.json
RUN npm install                     # Install dependencies
COPY . .                            # Copy all source code
RUN npm run build                   # Build optimized production files in /app/build

# ===== Stage 2: Run =====
FROM nginx:alpine                   # Use Nginx lightweight image for serving React
WORKDIR /usr/share/nginx/html       # Default Nginx HTML directory
COPY --from=builder /app/build .    # Copy React build files from builder
EXPOSE 80                           # Expose port 80 for web traffic
CMD ["nginx", "-g", "daemon off;"]  # Start Nginx in foreground
